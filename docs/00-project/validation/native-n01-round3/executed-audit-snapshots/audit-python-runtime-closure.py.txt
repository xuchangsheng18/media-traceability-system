import base64
import collections
import datetime
import hashlib
import importlib.metadata
import json
import os
import subprocess
from pathlib import Path
from packaging.requirements import Requirement
from packaging.utils import canonicalize_name
os.umask(0o077)
root=Path('/workspace/glwy/projects/media-traceability-system');prefix=Path('/opt/conda/glwy/envs/py310');out=Path('/workspace/glwy/runtime/media-traceability-system/native-python-closure-20261017');out.mkdir(mode=0o700,exist_ok=True)
distributions=list(importlib.metadata.distributions(path=[str(prefix/'lib/python3.10/site-packages')]))
versions={canonicalize_name(d.metadata['Name']):d.version for d in distributions}
report={'schema_version':1,'source_commit':subprocess.check_output(['git','rev-parse','HEAD'],cwd=root,text=True).strip(),'observed_at':datetime.datetime.now(datetime.timezone.utc).isoformat(),'scope':'active_marker_Python_dependency_resolution_and_installed_RECORD_integrity_not_upstream_wheel_signature_or_full_OS_closure','status':'NOT_EXECUTED','blockers':[],'distributions':[],'executed_script_sha256':hashlib.sha256(Path(__file__).read_bytes()).hexdigest(),'python_executable':str(prefix/'bin/python'),'files_checked':0,'hashed_files_checked':0,'record_hash_failures':[],'unresolved_requirements':[],'elf_files':[],'system_packages_changed':False,'original_runtime_lock_sha256':hashlib.sha256((root/'config/locks/requirements-py310.lock').read_bytes()).hexdigest()}
cache={}
for d in distributions:
    name=canonicalize_name(d.metadata['Name']);deps=[]
    for text in d.requires or []:
        req=Requirement(text)
        if req.marker and not req.marker.evaluate({'extra':''}):continue
        key=canonicalize_name(req.name);actual=versions.get(key);ok=actual is not None and (not req.specifier or req.specifier.contains(actual,prereleases=True))
        deps.append({'requirement':text,'resolved_version':actual,'satisfied':ok})
        if not ok:report['unresolved_requirements'].append({'package':name,'requirement':text,'resolved_version':actual})
    row={'name':name,'version':d.version,'requires_active':deps,'files_listed':len(d.files or []),'hashed_files':0,'unhashed_record_entries':0}
    for entry in d.files or []:
        path=Path(d.locate_file(entry)).resolve();report['files_checked']+=1
        if not path.is_relative_to(prefix):report['record_hash_failures'].append({'package':name,'reason':'RECORD_PATH_OUTSIDE_PREFIX','path':str(path)});continue
        if not path.is_file():
            if entry.hash:report['record_hash_failures'].append({'package':name,'reason':'MISSING_HASHED_FILE','path':str(path.relative_to(prefix))})
            continue
        if entry.hash:
            row['hashed_files']+=1;report['hashed_files_checked']+=1
            key=(str(path),entry.hash.mode)
            if key not in cache:
                digest=hashlib.new(entry.hash.mode)
                with path.open('rb') as stream:
                    first=True
                    for chunk in iter(lambda:stream.read(1024*1024),b''):
                        if first and chunk[:4]==b'\x7fELF':report['elf_files'].append(str(path))
                        first=False;digest.update(chunk)
                cache[key]=base64.urlsafe_b64encode(digest.digest()).rstrip(b'=').decode()
            if cache[key]!=entry.hash.value:report['record_hash_failures'].append({'package':name,'reason':'RECORD_DIGEST_MISMATCH','path':str(path.relative_to(prefix))})
        else:row['unhashed_record_entries']+=1
    report['distributions'].append(row)
conda=[]
for p in sorted((prefix/'conda-meta').glob('*.json')):
    d=json.loads(p.read_text());conda.append({'name':d['name'],'version':d['version'],'build':d['build'],'package_sha256':d.get('sha256'),'depends':d.get('depends',[]),'metadata_sha256':hashlib.sha256(p.read_bytes()).hexdigest(),'declared_license':d.get('license')})
report['conda_dependencies']=conda
if report['unresolved_requirements']:report['blockers'].append('UNRESOLVED_ACTIVE_PYTHON_REQUIREMENTS')
if report['record_hash_failures']:report['blockers'].append('INSTALLED_RECORD_INTEGRITY_FAILURES')
report['status']='FAIL' if report['blockers'] else 'PASS';report['elf_files']=sorted(set(report['elf_files']))
(out/'python-runtime-closure-report.json').write_text(json.dumps(report,indent=2)+'\n')
print(json.dumps({'status':report['status'],'python_distributions':len(distributions),'conda_packages':len(conda),'active_requirements':sum(len(d['requires_active']) for d in report['distributions']),'files_checked':report['files_checked'],'hashed_files':report['hashed_files_checked'],'integrity_failures':len(report['record_hash_failures']),'unresolved_requirements':len(report['unresolved_requirements']),'ELF_files':len(report['elf_files'])}),flush=True)
