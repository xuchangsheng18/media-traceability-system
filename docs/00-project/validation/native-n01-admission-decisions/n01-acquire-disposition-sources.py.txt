def fetch(kv):
 n,u=kv

  raw=urllib.request.urlopen(urllib.request.Request(u,headers={'User-Agent':'media-trace-N01-source-review'}),timeout=45).read();(o/n).write_bytes(raw)

   d=json.loads(raw);c=d['containers']['cna'];x.update(state=d['cveMetadata']['state'],description=[v['value'] for v in c.get('descriptions',[]) if v.get('lang')=='en'],affected=c.get('affected'),references=c.get('references'),tags=c.get('tags'))
